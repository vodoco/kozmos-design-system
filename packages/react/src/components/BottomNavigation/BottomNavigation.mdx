# BottomNavigation

Displays the BottomNavigation interface topology natively.

Its items are its own: an icon over an **11px label on a 14px line**, up to two lines, sharing the
bar's width, a selected item on the muted fill. They are not NavigationItem's rail items, and do not
take the rail's width, padding or selected bar. iOS draws the same 64pt tiles, and Android is
Material's navigation bar with 11sp labels.

package com.kozmos.components.bottomnavigation

import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Home
import androidx.compose.material.icons.filled.Notifications
import androidx.compose.material.icons.filled.Place
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue

@Composable
fun BottomNavigationExample() {
var route by remember { mutableStateOf("home") }
KozmosBottomNavigation(
items = listOf(
BottomNavigationItem(title = "Home", icon = Icons.Default.Home, route = "home"),
BottomNavigationItem(title = "Maps", icon = Icons.Default.Place, route = "maps"),
BottomNavigationItem(title = "Alerts", icon = Icons.Default.Notifications, route = "alerts")
),
currentRoute = route,
onNavigate = { route = it }
)
}
`}
