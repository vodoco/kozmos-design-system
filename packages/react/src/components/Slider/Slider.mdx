# Slider

Adjusts a numeric value or range. In React, provide `label`, `aria-label` or
`aria-labelledby` for a single thumb. For ranges, provide distinct, localized
`thumbLabels` (for example, minimum and maximum price). Helper/error descriptions
are associated with the interactive thumbs.

import SwiftUI
import Kozmos

// The slider draws no label: KozmosFieldWrapper puts one above it, and
// VoiceOver says the one the slider is given.
struct VolumeSlider: View {
    @Binding var volume: Double

    var body: some View {
        KozmosFieldWrapper(label: "Volume") {
            KozmosSlider(value: $volume, range: 0...100)
                .accessibilityLabel("Volume")
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.semantics.contentDescription
import androidx.compose.ui.semantics.semantics
import com.kozmos.components.fieldwrapper.KozmosFieldWrapper
import com.kozmos.components.slider.KozmosSlider

// The slider draws no label: KozmosFieldWrapper puts one above it, and
// TalkBack says the one the slider is given.
@Composable
fun VolumeSlider(volume: Float, onVolumeChange: (Float) -> Unit) {
    KozmosFieldWrapper(label = "Volume") {
        KozmosSlider(
            value = volume,
            onValueChange = onVolumeChange,
            valueRange = 0f..100f,
            modifier = Modifier.semantics { contentDescription = "Volume" }
        )
    }
}
`}
