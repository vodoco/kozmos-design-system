# Link

A styled anchor for navigation, with default and subtle colour treatments.
Supply `href` and meaningful link text; use Button for an action that does not navigate.

<PlatformSnippets
    react={`
import { Link } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Link href="/venues">Browse venues</Link>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosLink } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosLink />
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// SwiftUI's Link, in the theme colour: a press opens the URL through the
// environment's openURL, as an href does, and VoiceOver hears a link. There
// is no subtle variant.
struct BrowseVenuesLink: View {
    let venues: URL

    var body: some View {
        KozmosLink("Browse venues", destination: venues)
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.link.KozmosLink

// Underlined text in the theme colour, with no destination: a press calls
// onClick, and where it goes is yours. TalkBack is told no role, so it does
// not hear a link. There is no subtle variant.
@Composable
fun BrowseVenuesLink(showVenues: () -> Unit) {
    KozmosLink(text = "Browse venues", onClick = showVenues)
}
`}
