# Separator

A horizontal or vertical divider between related sections.
It is decorative by default; set `decorative={false}` when the separator should be exposed to assistive technology.

import SwiftUI
import Kozmos

// A line between two sections. Between the items of a row it is
// KozmosSeparator(orientation: .vertical).
struct PlaceSections: View {
    var body: some View {
        VStack(alignment: .leading, spacing: KozmosDimensions.primitivesLayoutSpacing150) {
            KozmosText("Opening hours", style: .headline)
            KozmosSeparator()

    kotlin={`
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column
import androidx.compose.runtime.Composable
import com.kozmos.components.separator.KozmosSeparator
import com.kozmos.components.text.KozmosText
import com.kozmos.tokens.KozmosDimensions

// A line between two sections. Between the items of a row it is
// KozmosSeparator(orientation = SeparatorOrientation.Vertical).
@Composable
fun PlaceSections() {
    Column(verticalArrangement = Arrangement.spacedBy(KozmosDimensions.primitivesLayoutSpacing150)) {
        KozmosText(text = "Opening hours")
        KozmosSeparator()
        KozmosText(text = "Monday–Friday, 09:00–18:00")
    }
}
`}
