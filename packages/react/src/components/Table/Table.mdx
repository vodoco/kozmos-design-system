# Table

Styled semantic table parts for headers, body rows, cells, footers and captions.
The wrapper supports horizontal overflow and keyboard scrolling; sorting, selection and data loading remain host responsibilities.

<Canvas>
  <Story of={TableStories.Default} />
</Canvas>

## Framed

`framed` draws a 1px border round the table and the control radius, 16, at its corners, as Figma, SwiftUI and
Compose draw a table (decision 159). The header and the rows are clipped inside the frame, so a row's hover and
selected fills follow its corners, and a caption sits inside it, at its foot. It is off by default, so a dense
page's table runs edge to edge and existing tables do not change.

import SwiftUI
import Kozmos

// The table draws a row for each element of your data, which is
// Identifiable, and a cell takes a String. It has no header row, header
// cell or caption: React's TableHeader, TableHead and TableCaption have no
// counterpart here. It is always framed, a border and the control radius,
// as React's Table is with framed.
struct PlacesTable: View {
    struct Place: Identifiable {
        let name: String
        let floor: String
        var id: String { name }
    }

    let places = [Place(name: "Reception", floor: "Ground")]

    var body: some View {
        KozmosTable(places) { place in
            KozmosTableRow {
                KozmosTableCell(place.name)
                KozmosTableCell(place.floor)
            }
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.table.KozmosTable
import com.kozmos.components.table.KozmosTableCell
import com.kozmos.components.table.KozmosTableHeaderCell
import com.kozmos.components.table.KozmosTableRow

// The header row goes in header, the body rows in the content. A cell takes
// a String, and the cells in a row share its width by weight, 1 each by
// default. There is no caption: React's TableCaption has no counterpart.
// It is always framed, a border and the control radius, as React's Table is
// with framed.
@Composable
fun PlacesTable() {
    KozmosTable(
        header = {
            KozmosTableRow {
                KozmosTableHeaderCell(text = "Name")
                KozmosTableHeaderCell(text = "Floor")
            }
        }
    ) {
        KozmosTableRow {
            KozmosTableCell(text = "Reception")
            KozmosTableCell(text = "Ground")
        }
    }
}
`}
