# Pagination

Composes page navigation from numbered links, previous/next controls and ellipses.
The host supplies destinations, the active page and pagination logic; the component does not fetch or slice data.

import SwiftUI
import Kozmos

// Each page is a button, not a link: there is no href, and where a press goes
// is yours. The current page is drawn, not told to VoiceOver, and Previous and
// Next take no words of yours.
struct ResultPages: View {
    let page: Int
    let goTo: (Int) -> Void

    var body: some View {
        KozmosPagination {
            KozmosPaginationLink(text: "1", isActive: page == 1) { goTo(1) }
            KozmosPaginationLink(text: "2", isActive: page == 2) { goTo(2) }
            KozmosPaginationNext { goTo(page + 1) }
        }
    }
}
`}

import androidx.compose.runtime.Composable
import com.kozmos.components.pagination.KozmosPagination
import com.kozmos.components.pagination.KozmosPaginationLink
import com.kozmos.components.pagination.KozmosPaginationNext

// Each page is pressable text, not a link: there is no href, and where a
// press goes is yours. TalkBack is told no role and not which page is
// current, and Previous and Next are fixed English words.
@Composable
fun ResultPages(page: Int, goTo: (Int) -> Unit) {
    KozmosPagination {
        KozmosPaginationLink(text = "1", isActive = page == 1, onClick = { goTo(1) })
        KozmosPaginationLink(text = "2", isActive = page == 2, onClick = { goTo(2) })
        KozmosPaginationNext(onClick = { goTo(page + 1) })
    }
}
`}
