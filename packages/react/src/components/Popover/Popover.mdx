# Popover

An anchored surface for supporting content or small interactions, composed from a trigger and content.
An optional PopoverArrow connects it visually to its anchor; owned portals retain the enclosing Kozmos theme.

    swift={`
import SwiftUI
import Kozmos

// The popover Kozmos draws is a title and an optional description. Given
// content of its own instead, it draws only that content.
struct OpeningHoursPopover: View {
    @State private var isPresented = false

    var body: some View {
        KozmosButton("Opening hours") { isPresented = true }
            .kozmosPopover(
                isPresented: $isPresented,
                title: "Opening hours",
                description: "Monday–Friday, 09:00–18:00"
            )
    }
}
`}
    kotlin={`
import androidx.compose.foundation.layout.Box
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.popover.KozmosPopover
import com.kozmos.components.text.KozmosText

// Compose's popover opens over its parent's top-start corner, so it shares a
// Box with the button that opens it.
@Composable
fun OpeningHoursPopover() {
    var isExpanded by remember { mutableStateOf(false) }
    Box {
        KozmosButton(onClick = { isExpanded = true }) { Text("Opening hours") }
        KozmosPopover(
            isExpanded = isExpanded,
            onDismissRequest = { isExpanded = false }
        ) {
            KozmosText(text = "Monday–Friday, 09:00–18:00")
        }
    }
}
`}
