# Grid

Arranges children in a CSS grid with explicit column, row, flow, alignment and gap options.
Use it for two-dimensional layouts; use Stack for a row or column of content.

import { Grid } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Grid>

import { KozmosGrid } from "@kozmos-ds/vue"

  <KozmosGrid />
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// Equal columns, two unless cols says otherwise, 16pt apart both ways; what
// it holds fills them in order. There are no row, flow or alignment options.
// Kozmos's Grid shares SwiftUI's name, and with Kozmos imported, Grid is
// Kozmos's.
struct VenueAreas: View {
    var body: some View {
        Grid(cols: 2) {
            KozmosText("Reception")
            KozmosText("Meeting rooms")
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.grid.KozmosGrid
import com.kozmos.components.text.KozmosText

// Equal columns, two unless cols says otherwise, 16dp apart both ways. It is
// a LazyVerticalGrid: what it holds is item { } calls, and it scrolls itself,
// so it cannot sit in a parent that scrolls vertically. There are no row,
// flow or alignment options.
@Composable
fun VenueAreas() {
    KozmosGrid(cols = 2) {
        item { KozmosText(text = "Reception") }
        item { KozmosText(text = "Meeting rooms") }
    }
}
`}
