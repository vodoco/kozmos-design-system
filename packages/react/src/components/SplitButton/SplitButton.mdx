# SplitButton

Pairs a primary action with a separate dropdown of related actions.
Supply `onMainClick` for the primary button and `menuItems` for the alternatives; `disabled` applies to both triggers.

Both halves are the theme fill, and each draws the pressed token, `#0D44C2`, while it is held, on
the web and in Compose. On iOS the menu half keeps SwiftUI's own press: it is a `Menu`, which does
not run a custom button style's pressed state, and drawing the token there would mean replacing the
system menu. This is a deliberate platform difference (decision 64); the action half draws the token
on iOS too.

import { useState } from "react";
import { SplitButton } from "@kozmos-ds/react";
export function Example() {
  const [action, setAction] = useState("None");

      <SplitButton
        onMainClick={() => setAction("Publish requested")}

}

    swift={`
import SwiftUI
import Kozmos

// The label is a String, and each menu item a title and what it does; the
// menu half is the system's Menu. There is no disabled state.
struct PublishOrSaveDraft: View {
    let publish: () -> Void
    let saveDraft: () -> Void

    var body: some View {
        KozmosSplitButton(
            label: "Publish",
            mainAction: publish,
            menuItems: [("Save draft", saveDraft)]
        )
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.splitbutton.KozmosSplitButton
import com.kozmos.components.splitbutton.SplitContextMenuItem

// The label is a String, and each menu item a SplitContextMenuItem: its
// label and what it does. There is no disabled state, and TalkBack names the
// menu half "More actions", in fixed English.
@Composable
fun PublishOrSaveDraft(publish: () -> Unit, saveDraft: () -> Unit) {
    KozmosSplitButton(
        label = "Publish",
        onMainClick = publish,
        menuItems = listOf(SplitContextMenuItem(label = "Save draft", onClick = saveDraft))
    )
}
`}
