    swift={`
import SwiftUI
import Kozmos

// The initials show while the image loads, if it fails, and with no URL.
// The avatar takes no alt text: the label is what VoiceOver says.
struct ProfileAvatar: View {
    var body: some View {
        KozmosAvatar(
            imageURL: URL(string: "https://github.com/shadcn.png"),
            fallbackText: "CN"
        )
        .accessibilityLabel("@shadcn")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.semantics.clearAndSetSemantics
import androidx.compose.ui.semantics.contentDescription
import com.kozmos.components.avatar.KozmosAvatar

// The initials show while the image loads, if it fails, and with no URL.
// The avatar takes no alt text: TalkBack says the description it is given.
@Composable
fun ProfileAvatar() {
    KozmosAvatar(
        imageUrl = "https://github.com/shadcn.png",
        fallbackText = "CN",
        modifier = Modifier.clearAndSetSemantics { contentDescription = "@shadcn" }
    )
}
`}
