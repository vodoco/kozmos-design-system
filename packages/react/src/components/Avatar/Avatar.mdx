    swift={`
import SwiftUI

struct KozmosAvatar: View {
    let imageURL: URL?
    let fallbackText: String
    let size: CGFloat = 40
    
    var body: some View {
        ZStack {
            // Fallback layer
            Circle()
                .fill(KozmosColors.semanticColorBackgroundMuted)
            Text(fallbackText)
                .font(.system(size: size * 0.4, weight: .medium))
                .foregroundColor(KozmosColors.semanticColorTextMuted)
            
            // Image layer
            if let url = imageURL {
                AsyncImage(url: url) { phase in
                    if let image = phase.image {
                        image
                            .resizable()
                            .aspectRatio(contentMode: .fill)
                            .clipShape(Circle())
                    }
                }
            }
        }
        .frame(width: size, height: size)
        .overlay(
             Circle().stroke(KozmosColors.semanticColorBorderDefault, lineWidth: 1)
        )
    }
}
    `}
    kotlin={`
package com.kozmos.components.avatar

import com.kozmos.tokens.KozmosDimensions

import androidx.compose.foundation.background
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.size
import androidx.compose.foundation.shape.CircleShape
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.unit.dp
import coil.compose.AsyncImage
import com.kozmos.tokens.KozmosColors

@Composable
fun KozmosAvatar(
    imageUrl: String?,
    fallbackText: String,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .size(KozmosDimensions.primitivesLayoutSizing500)
            .clip(CircleShape)
            .background(KozmosColors.primitivesColorsBackground300),
        contentAlignment = Alignment.Center
    ) {
        if (imageUrl != null) {
            AsyncImage(
                model = imageUrl,
                contentDescription = null,
                contentScale = ContentScale.Crop,
                modifier = Modifier.matchParentSize()
            )
        } else {
            Text(
                text = fallbackText,
                style = MaterialTheme.typography.titleMedium,
                color = KozmosColors.primitivesColorsForeground500
            )
        }
    }
}

    `}
