## Sizes

`size="sm"` draws the Avatar at 24, its initials at 12/16, where `default` is 40 (decision 221): for a row that keeps
its own height, as an option's media in Combobox, Listbox and MultiSelect does (decision 215). SwiftUI's
`KozmosAvatar(size: .sm)` and Compose's `KozmosAvatar(size = KozmosAvatarSize.Sm)` draw the same, and Figma's Avatar
set has a `Size` of `Default` and `Sm`.

<PlatformSnippets
    react={`
import { Avatar, AvatarFallback, AvatarImage } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Avatar>
        <AvatarImage src="https://github.com/shadcn.png" alt="@shadcn" />
        <AvatarFallback>CN</AvatarFallback>
      </Avatar>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosAvatar, KozmosAvatarFallback, KozmosAvatarImage } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosAvatar>
    <KozmosAvatarImage src="https://github.com/shadcn.png" alt="@shadcn" />
    <KozmosAvatarFallback>CN</KozmosAvatarFallback>
  </KozmosAvatar>
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// The initials show while the image loads, if it fails, and with no URL.
// The avatar takes no alt text: the label is what VoiceOver says.
struct ProfileAvatar: View {
    var body: some View {
        KozmosAvatar(
            imageURL: URL(string: "https://github.com/shadcn.png"),
            fallbackText: "CN"
        )
        .accessibilityLabel("@shadcn")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.semantics.clearAndSetSemantics
import androidx.compose.ui.semantics.contentDescription
import com.kozmos.components.avatar.KozmosAvatar

// The initials show while the image loads, if it fails, and with no URL.
// The avatar takes no alt text: TalkBack says the description it is given.
@Composable
fun ProfileAvatar() {
    KozmosAvatar(
        imageUrl = "https://github.com/shadcn.png",
        fallbackText = "CN",
        modifier = Modifier.clearAndSetSemantics { contentDescription = "@shadcn" }
    )
}
`}
