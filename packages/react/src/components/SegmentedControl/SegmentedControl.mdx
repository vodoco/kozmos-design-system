# SegmentedControl

Use SegmentedControl when users choose one option from a short, visible set.
Prefer Radio for longer lists and Tabs when the selected item changes visible
content regions.

<PlatformSnippets
  react={`
import { SegmentedControl } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <SegmentedControl
        defaultValue="overview"
        items={[
          { value: "overview", label: "Overview" },
          { value: "details", label: "Details" },
          { value: "activity", label: "Activity" },
        ]}
      />
    </>
  );
}
  `}
  vue={`
<script setup>
import { KozmosSegmentedControl } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosSegmentedControl />
</template>
  `}
  swift={`
@State private var selection = 0

KozmosSegmentedControl(
selection: $selection,
items: ["Overview", "Details", "Activity"]
)
`}
  kotlin={`
var selectedIndex by remember { mutableIntStateOf(0) }

KozmosSegmentedControl(
options = listOf("Overview", "Details", "Activity"),
selectedIndex = selectedIndex,
onOptionSelected = { selectedIndex = it }
)
`}
/>
