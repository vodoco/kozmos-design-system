# Stack

Arranges children in a flex row or column with standard gap, alignment, justification and wrapping options.
Use it for one-dimensional composition without repeating layout classes.

<PlatformSnippets
    react={`
import { Stack } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Stack>
        <p>Reception</p>
        <p>Ground floor</p>
      </Stack>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosStack } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosStack />
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// A column with Kozmos's spacing between its children, 8pt by default.
// Unlike React's column it never stretches them: .start lines them up on the
// leading edge. There is no wrap or justify.
struct ReceptionStack: View {
    var body: some View {
        Stack(alignment: .start) {
            KozmosText("Reception")
            KozmosText("Ground floor")
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.stack.KozmosStack
import com.kozmos.components.text.KozmosText

// A column with Kozmos's spacing between its children, 8dp by default, on
// its start edge. It takes no alignment, wrap or justify, and its children
// are in no ColumnScope, so none can take a weight.
@Composable
fun ReceptionStack() {
    KozmosStack {
        KozmosText(text = "Reception")
        KozmosText(text = "Ground floor")
    }
}
`}
