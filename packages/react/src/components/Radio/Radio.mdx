# Radio Group

A set of checkable buttons—known as radio buttons—where no more than one of the buttons can be checked at a time.

## Usage

Use radio buttons when:
- Users need to select a single option from a list.
- All available options should be visible at once (unlike Select).
- There are fewer than 5-7 options.

    swift={`
import SwiftUI
import Kozmos

// Each option draws its own label, and every option shares your selection:
// the group only stacks them, and takes no name (React's aria-label has no
// counterpart). VoiceOver is not told which option is selected.
struct ChooseAnOption: View {
    @State private var selection = "option-one"

    var body: some View {
        KozmosRadioGroup {
            KozmosRadioGroupItem(value: "option-one", label: "Option One", selection: $selection)
            KozmosRadioGroupItem(value: "option-two", label: "Option Two", selection: $selection)
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import com.kozmos.components.radio.KozmosRadioGroup
import com.kozmos.components.radio.KozmosRadioGroupItem

// Each option draws its own label, and every option shares your selection:
// the group stacks them as one selectable group, and takes no name (React's
// aria-label has no counterpart).
@Composable
fun ChooseAnOption() {
    var selection by remember { mutableStateOf("option-one") }
    KozmosRadioGroup {
        KozmosRadioGroupItem(
            value = "option-one",
            selectedValue = selection,
            onOptionSelected = { selection = it },
            label = "Option One"
        )
        KozmosRadioGroupItem(
            value = "option-two",
            selectedValue = selection,
            onOptionSelected = { selection = it },
            label = "Option Two"
        )
    }
}
`}
