# IconButton

IconButtons are used for actions that don't require a label, such as "flaging" or "liking" an item.

The React component uses the shared Button control radius (16px), not a circular
pill override. Its minimum hit target remains 44×44px. Always provide an
accessible label for the action.

import SwiftUI

public struct KozmosIconButton: View {
let iconName: String
let action: () -> Void

    public init(
        iconName: String,
        action: @escaping () -> Void
    ) {
        self.iconName = iconName
        self.action = action
    }

    public var body: some View {
        Button(action: action) {
            Image(systemName: iconName)
                .font(.system(size: 20))
                .foregroundColor(.primary)
                .padding(8)
                .background(Color.clear)
                .clipShape(Circle())
        }
    }

}
`}

import androidx.compose.material3.IconButton
import androidx.compose.material3.Icon
import androidx.compose.runtime.Composable
import androidx.compose.ui.graphics.vector.ImageVector

@Composable
fun KozmosIconButton(
icon: ImageVector,
onClick: () -> Unit,
contentDescription: String? = null
) {
IconButton(onClick = onClick) {
Icon(
imageVector = icon,
contentDescription = contentDescription
)
}
}
