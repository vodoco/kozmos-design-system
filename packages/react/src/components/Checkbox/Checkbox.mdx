# Checkbox

A control that allows the user to toggle between checked and unchecked states.

## Usage

Use checkboxes when:
- Users need to select one or more options from a list.
- Users need to toggle a single setting (e.g., "Accept Terms").

## Implementation

import { Checkbox } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <div className="flex items-center space-x-2">
        <Checkbox id="terms" />
        <label
          htmlFor="terms"
          className="text-sm font-medium leading-none peer-disabled:cursor-not-allowed peer-disabled:opacity-70"
        >
          Accept terms and conditions

      </div>
    </>
  );
}

    vue={`
<script setup>
import { KozmosCheckbox } from "@kozmos-ds/vue"
import { ref } from "vue"

const isChecked = ref(false)
</script>

<template>
  <div class="flex items-center space-x-2">
    <KozmosCheckbox id="terms" v-model:checked="isChecked" />
    <label
      for="terms"
      class="text-sm font-medium leading-none peer-disabled:cursor-not-allowed peer-disabled:opacity-70"
    >
      Accept terms and conditions
    </label>
  </div>
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// The checkbox draws its label beside it, and the whole row toggles. It has
// two states, checked and not: no indeterminate one.
struct AcceptTerms: View {
    @Binding var accepted: Bool

    var body: some View {
        KozmosCheckbox(checked: $accepted, label: "Accept terms and conditions")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.checkbox.KozmosCheckbox

// The checkbox draws its label beside it, and the whole row toggles. It has
// two states, checked and not: no indeterminate one.
@Composable
fun AcceptTerms(accepted: Boolean, onAcceptedChange: (Boolean) -> Unit) {
    KozmosCheckbox(
        checked = accepted,
        onCheckedChange = onAcceptedChange,
        label = "Accept terms and conditions"
    )
}
`}
/>
