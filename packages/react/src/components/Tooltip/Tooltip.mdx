import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <TooltipProvider>
        <Tooltip>
          <TooltipTrigger>Hover</TooltipTrigger>
          <TooltipContent>
            <p>Add to library</p>
          </TooltipContent>
        </Tooltip>
      </TooltipProvider>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosButton } from "@kozmos-ds/vue"
// @kozmos-ds/vue cannot attach a tooltip yet: it has no TooltipProvider, TooltipTrigger or TooltipContent, and KozmosTooltip throws outside a provider.
</script>

<template>
  <KozmosButton>Hover</KozmosButton>
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// The bubble shows on hover, or when a finger holds the view down. Its text
// is VoiceOver's hint too; pass isAccessibilityHint: false where it repeats
// the view's own label.
struct AddToLibrary: View {
    let add: () -> Void

    var body: some View {
        KozmosButton("Hover", action: add)
            .kozmosTooltip("Add to library")
    }
}
`}
    kotlin={`
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.interaction.collectIsPressedAsState
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.tooltip.KozmosTooltip

// Compose's tooltip shows only while visible is true: it watches nothing
// itself, so this one follows the button's press. Its side is Top by
// default; Top, Right, Bottom and Left are physical, as React's are, and
// Leading and Trailing follow the reading direction.
@Composable
fun AddToLibrary(add: () -> Unit) {
    val interactions = remember { MutableInteractionSource() }
    val pressed by interactions.collectIsPressedAsState()
    KozmosTooltip(tooltip = "Add to library", visible = pressed) {
        KozmosButton(onClick = add, interactionSource = interactions) { Text("Hover") }
    }
}
`}
