# Tooltip

A popup that displays information related to an element when the element receives keyboard focus or the mouse hovers over it.

<Canvas>
  <Story of={TooltipStories.Default} />
</Canvas>

<Controls />

## Sides

`side` names the side of its trigger the tooltip opens on: top, right, bottom or left, the trigger's own sides in every language, as Radix's `side` and Figma's Side property are. In a right-to-left layout `side="right"` still opens on the trigger's right. SwiftUI also takes `.leading` and `.trailing`, which follow the reading direction: `.trailing` is the trigger's right left to right and its left right to left. The tip points at the trigger on every side.

import SwiftUI
import Kozmos

// The bubble shows on hover, or when a finger holds the view down. Its text
// is VoiceOver's hint too; pass isAccessibilityHint: false where it repeats
// the view's own label. side: .right is the view's right in every language,
// as React's side is; .trailing is after it in the reading direction.
struct AddToLibrary: View {
    let add: () -> Void

    var body: some View {

    }

import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.interaction.collectIsPressedAsState
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.tooltip.KozmosTooltip

// Compose's tooltip shows only while visible is true: it watches nothing
// itself, so this one follows the button's press. Its side is Top by
// default; Top, Right, Bottom and Left are physical, as React's are, and
// Leading and Trailing follow the reading direction.
@Composable
fun AddToLibrary(add: () -> Unit) {
    val interactions = remember { MutableInteractionSource() }
    val pressed by interactions.collectIsPressedAsState()
    KozmosTooltip(tooltip = "Add to library", visible = pressed) {
        KozmosButton(onClick = add, interactionSource = interactions) { Text("Hover") }
    }
}
`}

## Inverse tone

`tone="inverse"` on TooltipContent draws the dark tooltip of Pointr Cloud's info icons and its
guidance over the map (decision 192, GAP-210): the inverse surface, `Semantics.Inverse`, under its
words, and the arrow in the surface's colour. In light it is foreground/200, #2E3138, under white,
13.0:1, exactly Cloud's tooltip; in dark it turns over, #D1CEC7 under black, 13.4:1. The roles
follow a client's theme. It is the same size as the default, whose border it keeps in the
surface's colour. The default tone is unchanged: the popover surface with its border.

import SwiftUI
import Kozmos

// tone: .inverse draws the dark bubble, Semantics.Inverse: dark on a light
// page and light on a dark one, the same size as the default bubble.
struct SelectionHint: View {
    let adjust: () -> Void

    var body: some View {
        KozmosButton("Adjust", variant: .outline, action: adjust)
            .kozmosTooltip("Keep holding Shift for multi-selection", tone: .inverse)
    }
}
`}
    kotlin={`
import androidx.compose.foundation.interaction.MutableInteractionSource
import androidx.compose.foundation.interaction.collectIsPressedAsState
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.remember
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.button.KozmosButtonVariant
import com.kozmos.components.tooltip.KozmosTooltip
import com.kozmos.components.tooltip.KozmosTooltipTone

// The inverse tone is the dark tooltip: the inverse surface under the
// inverse on-surface's words, in both themes, the same size as the default.
@Composable
fun ShiftToMultiSelect(adjust: () -> Unit) {
    val interactions = remember { MutableInteractionSource() }
    val pressed by interactions.collectIsPressedAsState()
    KozmosTooltip(
        tooltip = "Keep holding Shift for multi-selection",
        visible = pressed,
        tone = KozmosTooltipTone.Inverse
    ) {
        KozmosButton(onClick = adjust, variant = KozmosButtonVariant.Outline, interactionSource = interactions) {
            Text("Adjust")
        }
    }
}
`}
/>
