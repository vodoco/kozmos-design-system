# Breadcrumb

Composes a navigation trail from links, separators and a non-interactive current-page label.
Use the exported parts to reflect the page hierarchy; it does not infer routes or generate the trail.

import SwiftUI
import Kozmos

// The trail is its titles, in order, and the last is the current page,
// which cannot be pressed. A press hands back the item's index: where it
// goes is yours, as the trail has no links.
struct VenueTrail: View {
    let showVenues: () -> Void

    var body: some View {
        KozmosBreadcrumb(items: ["Venues", "Central venue"], onSelect: { index in
            if index == 0 { showVenues() }
        })
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.breadcrumb.KozmosBreadcrumb

// The trail is its titles, in order, and the last is the current page,
// which cannot be pressed. A press hands back the item's index: where it
// goes is yours, as the trail has no links.
@Composable
fun VenueTrail(showVenues: () -> Unit) {
    KozmosBreadcrumb(
        items = listOf("Venues", "Central venue"),
        onItemClick = { index -> if (index == 0) showVenues() }
    )
}
