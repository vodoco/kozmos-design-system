<PlatformSnippets
    react={`
import { Switch } from "@kozmos-ds/react";
import { Label } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <div className="flex items-center space-x-2">
        <Switch id="airplane-mode" />
        <Label htmlFor="airplane-mode">Airplane Mode</Label>
      </div>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosSwitch, KozmosLabel } from "@kozmos-ds/vue"
import { ref } from "vue"

const isChecked = ref(false)
</script>

<template>
  <div class="flex items-center space-x-2">
    <KozmosSwitch id="airplane-mode" v-model:checked="isChecked" />
    <KozmosLabel htmlFor="airplane-mode">Airplane Mode</KozmosLabel>
  </div>
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// The switch draws its label beside it, and the whole row toggles.
struct AirplaneMode: View {
    @Binding var isOn: Bool

    var body: some View {
        KozmosSwitch(checked: $isOn, label: "Airplane Mode")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.switch.KozmosSwitch

// The switch draws its label beside it, and the whole row toggles.
@Composable
fun AirplaneMode(isOn: Boolean, onChange: (Boolean) -> Unit) {
    KozmosSwitch(checked = isOn, onCheckedChange = onChange, label = "Airplane Mode")
}
`}
/>
