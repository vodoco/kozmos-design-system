## Implementation

The close shows only a cross and is named "Dismiss notification" on all three platforms. A product
that translates passes its own words: `aria-label` on React's `ToastClose`, `closeLabel` on
SwiftUI's and Compose's `KozmosToast` (and on Compose's `KozmosToast(hostState)`, for every
snackbar it draws).

import { useState } from "react";
import {
  Button,
  Toast,
  ToastClose,
  ToastDescription,
  ToastProvider,
  ToastTitle,
  ToastViewport,
} from "@kozmos-ds/react";
export function Example() {
  const [open, setOpen] = useState(false);
  return (
    <ToastProvider>
      <Button onClick={() => setOpen(true)}>Show toast</Button>
      <Toast open={open} onOpenChange={setOpen}>
        <div>
          <ToastTitle>Location saved</ToastTitle>
          <ToastDescription>Find it in your saved places.</ToastDescription>
        </div>
        <ToastClose aria-label="Dismiss notification" />
      </Toast>
      <ToastViewport />
    </ToastProvider>
  );
}

    vue={`
<script setup>
import { KozmosButton } from "@kozmos-ds/vue"
// @kozmos-ds/vue cannot show a toast yet: it has no ToastProvider, ToastViewport, ToastTitle, ToastDescription or ToastClose and no toast(), and KozmosToast throws outside a provider.
</script>

<template>
  <KozmosButton>Show toast</KozmosButton>
</template>

    swift={`
import SwiftUI
import Kozmos

// The toast draws itself at the bottom of the space it is given, so it goes
// over the screen's content. It closes after 3 seconds (autoDismissAfter) or
// at its close button, named by closeLabel, "Dismiss notification" unless you
// pass your own.
struct LocationSavedToast: View {
    @State private var isPresented = false

    var body: some View {
        ZStack {
            KozmosButton("Show toast") { isPresented = true }
            KozmosToast(
                isPresented: $isPresented,
                title: "Location saved",
                description: "Find it in your saved places."
            )
        }
    }
}
`}
    kotlin={`
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.toast.KozmosToast
import com.kozmos.tokens.KozmosDimensions

// This KozmosToast draws where you put it and stays until its close button,
// named by closeLabel ("Dismiss notification" unless you pass your own),
// calls onDismiss: it does not time out.
// KozmosToast(hostState) draws a SnackbarHostState's snackbars the same way,
// with a message, an action and a close button, but no description.
@Composable
fun LocationSavedToast() {
    var isOpen by remember { mutableStateOf(false) }
    Box(modifier = Modifier.fillMaxSize()) {
        KozmosButton(onClick = { isOpen = true }) { Text("Show toast") }
        if (isOpen) {
            KozmosToast(
                title = "Location saved",
                description = "Find it in your saved places.",
                modifier = Modifier
                    .align(Alignment.BottomCenter)
                    .padding(KozmosDimensions.primitivesLayoutSpacing300),
                onDismiss = { isOpen = false }
            )
        }
    }
}
`}
