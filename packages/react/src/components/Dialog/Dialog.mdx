import SwiftUI

// Usage example modifier
extension View {
    func kozmosDialog(isPresented: Binding<Bool>, title: String, message: String) -> some View {
        self.alert(isPresented: isPresented) {
            Alert(
                title: Text(title).foregroundColor(KozmosColors.semanticColorTextDefault),
                message: Text(message).foregroundColor(KozmosColors.semanticColorTextMuted),
                dismissButton: .default(Text("OK"))
            )
        }
    }
}

// Or Custom View
struct KozmosDialogView<Content: View>: View {
    let title: String
    let description: String
    let content: Content
    
    var body: some View {
        VStack(spacing: 16) {
            Text(title)
                .font(.headline)
                .foregroundColor(KozmosColors.semanticColorTextDefault)
            Text(description)
                .font(.subheadline)
                .foregroundColor(KozmosColors.semanticColorTextMuted)
                .multilineTextAlignment(.center)
            
            content
        }
        .padding()
        .background(KozmosColors.semanticColorSurfaceDefault)
        .cornerRadius(12)
        .shadow(color: Color.black.opacity(0.1), radius: 10, x: 0, y: 4)
        .padding()
    }
}

package com.kozmos.components.dialog

import com.kozmos.tokens.KozmosColors
import com.kozmos.tokens.KozmosDimensions

import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.padding
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.MaterialTheme
import androidx.compose.material3.Surface
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.compose.ui.window.Dialog

@Composable
fun KozmosDialog(
    onDismissRequest: () -> Unit,
    modifier: Modifier = Modifier,
    content: @Composable () -> Unit
) {
    Dialog(onDismissRequest = onDismissRequest) {
        Surface(
            shape = RoundedCornerShape(KozmosDimensions.primitivesLayoutRadius200),
            color = MaterialTheme.colorScheme.surface,
            modifier = modifier
        ) {
            Box(modifier = Modifier.padding(KozmosDimensions.primitivesLayoutSpacing300)) {
                content()
            }
        }
    }
}
