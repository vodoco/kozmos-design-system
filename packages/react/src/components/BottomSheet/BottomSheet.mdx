# BottomSheet

A bottom-edge Drawer composition with a handle, optional close button and safe-area-aware spacing.
Provide `title` for a visually hidden accessible name or render BottomSheetTitle; use AdaptiveMapShell for a persistent map panel with detents.

import SwiftUI
import Kozmos

// The system's sheet, with its drag indicator, at the medium and large
// detents; a swipe down closes it. It draws no title, description, close
// button or padding of its own: the content pads itself.
struct LocationDetailsSheet: View {
    @State private var isPresented = false

    var body: some View {
        KozmosButton("Show details") { isPresented = true }
            .kozmosBottomSheet(isPresented: $isPresented) {
                VStack(alignment: .leading, spacing: KozmosDimensions.primitivesLayoutSpacing100) {
                    KozmosText("Location details", style: .headline)
                    KozmosText("Reception is on the ground floor.", tone: .muted)
                }
                .padding(KozmosDimensions.primitivesLayoutSpacing300)
            }
    }
}
`}
    kotlin={`
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.padding
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import androidx.compose.ui.Modifier
import com.kozmos.components.bottomsheet.KozmosBottomSheet
import com.kozmos.components.button.KozmosButton
import com.kozmos.components.text.KozmosText
import com.kozmos.tokens.KozmosDimensions

// The sheet is open while it is composed; a swipe down or a tap outside asks
// for it to close. It draws a drag handle, and no title, description, close
// button or padding of its own: the content pads itself. Compose's
// KozmosText takes no style or tone, so the title reads as body text.
@Composable
fun LocationDetailsSheet() {
    var isOpen by remember { mutableStateOf(false) }
    KozmosButton(onClick = { isOpen = true }) { Text("Show details") }
    if (isOpen) {
        KozmosBottomSheet(onDismissRequest = { isOpen = false }) {
            Column(
                modifier = Modifier.padding(KozmosDimensions.primitivesLayoutSpacing300),
                verticalArrangement = Arrangement.spacedBy(KozmosDimensions.primitivesLayoutSpacing100)
            ) {
                KozmosText(text = "Location details")
                KozmosText(text = "Reception is on the ground floor.")
            }
        }
    }
}
`}
