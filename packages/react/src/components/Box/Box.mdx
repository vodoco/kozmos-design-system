# Box

A minimal layout wrapper that renders a `div`, or passes its props to one child with `asChild`.
It adds no spacing or surface styling of its own; use Stack, Grid or Container for those layout conventions.

<PlatformSnippets
    react={`
import { Box } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Box>
        <p>Venue overview</p>
      </Box>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosBox } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosBox />
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// SwiftUI's Box is named Box, with no Kozmos prefix. It adds nothing, not
// even a stack: what it holds lays out as its parent lays it out.
struct OverviewBox: View {
    var body: some View {
        Box {
            KozmosText("Venue overview")
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.box.KozmosBox
import com.kozmos.components.text.KozmosText

// KozmosBox is Compose's Box and adds nothing to it: what it holds stacks
// at its top-start corner, one child over the next.
@Composable
fun OverviewBox() {
    KozmosBox {
        KozmosText(text = "Venue overview")
    }
}
`}
