# List

A bordered vertical list with standard or compact row density.
Compose ListItem children for content rows; use POIResultList for SDK search results and their selection/action contract.

    swift={`
import SwiftUI
import Kozmos

// SwiftUI's List, in its plain style: it scrolls by itself, fills the height
// it is given and draws the system's separators. A row is its text, and
// there is no border or compact density.
struct PlaceList: View {
    var body: some View {
        KozmosList {
            KozmosListItem("Reception")
            KozmosListItem("Atrium Café")
        }
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.list.KozmosList
import com.kozmos.components.list.KozmosListItem

// A Column of rows, each its text with a divider under it, which the last
// row turns off here. There is no border or compact density, and TalkBack is
// not told it is a list.
@Composable
fun PlaceList() {
    KozmosList {
        KozmosListItem(text = "Reception")
        KozmosListItem(text = "Atrium Café", showDivider = false)
    }
}
`}
