# Rating

Collects or displays a rating using stars or a thumbs-up/down choice.
Use `value` and `onChange` for input, `readOnly` for display, and the label props to provide localized accessible names.

    swift={`
import SwiftUI
import Kozmos

// 0 is unrated, and choosing the chosen star clears it. label is the
// group's name for VoiceOver, as aria-label is on the web.
struct RateYourVisit: View {
    @State private var rating = 0

    var body: some View {
        KozmosRating(value: $rating, label: "Rate your visit")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableIntStateOf
import androidx.compose.runtime.remember
import androidx.compose.runtime.setValue
import com.kozmos.components.rating.KozmosRating

// 0 is unrated, and choosing the chosen star clears it. label is the
// group's name for TalkBack, as aria-label is on the web.
@Composable
fun RateYourVisit() {
    var rating by remember { mutableIntStateOf(0) }
    KozmosRating(value = rating, onValueChange = { rating = it }, label = "Rate your visit")
}
`}
