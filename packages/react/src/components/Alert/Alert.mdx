import SwiftUI
import Kozmos

// KozmosAlertIcon draws its variant's symbol. The alert's own variant
// colours only its edge, so a warning passes .warning to both. It
// announces nothing: it is part of the page, as React's is by default.
struct ClosingTimeAlert: View {
    var body: some View {
        KozmosAlert {
            KozmosAlertIcon()
            KozmosAlertContent {
                KozmosAlertTitle("Heads up!")
                KozmosAlertDescription("The venue closes at 18:00 today.")
            }
        }
    }
}
`}

import androidx.compose.runtime.Composable
import com.kozmos.components.alert.KozmosAlert
import com.kozmos.components.alert.KozmosAlertContent
import com.kozmos.components.alert.KozmosAlertDescription
import com.kozmos.components.alert.KozmosAlertIcon
import com.kozmos.components.alert.KozmosAlertTitle

// KozmosAlertIcon draws its status's icon. The alert's own status colours
// only its edge, so a warning passes AlertStatus.Warning to both. It
// announces nothing: it is part of the page, as React's is by default.
@Composable
fun ClosingTimeAlert() {
    KozmosAlert {
        KozmosAlertIcon()
        KozmosAlertContent {
            KozmosAlertTitle(title = "Heads up!")
            KozmosAlertDescription(description = "The venue closes at 18:00 today.")
        }
    }
}
