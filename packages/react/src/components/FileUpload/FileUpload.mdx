# FileUpload

A file-selection surface with browsing, drag-and-drop, selected-file removal and validation feedback.
Use its callbacks to receive files and perform uploads in the host; selecting a file does not upload it.

import SwiftUI
import Kozmos

// One file, from the system's importer, of the types allowedContentTypes
// names (any, unless it says otherwise): no drag and drop, no second file, no
// size limit and no message. The part draws no label, so KozmosFieldWrapper
// puts one above it. Picking a file hands you its URL and uploads nothing;
// removing it tells you nothing.
struct DocumentUpload: View {
    let receive: (URL) -> Void

    var body: some View {
        KozmosFieldWrapper(label: "Documents") {
            KozmosFileUpload(onFileSelected: receive)
        }
    }
}
`}
    kotlin={`
import android.net.Uri
import androidx.compose.runtime.Composable
import com.kozmos.components.fieldwrapper.KozmosFieldWrapper
import com.kozmos.components.fileupload.KozmosFileUpload

// One file of any type, picked with the system's picker: no drag and drop, no
// second file, no type or size limit and no message. The part draws no label,
// so KozmosFieldWrapper puts one above it. Picking a file hands you its Uri
// and uploads nothing; removing it tells you nothing.
@Composable
fun DocumentUpload(receive: (Uri) -> Unit) {
    KozmosFieldWrapper(label = "Documents") {
        KozmosFileUpload(onFileSelected = receive)
    }
}
`}
