## Variants

`vertical-list`, `horizontal-list` and `compact-stepper` exist on every platform. iOS also has
`collapsible`, for a control parked in a corner of a map: it shows the current level only and
opens the full list on touch, each row the level's short label beside its name, the current one
filled. React and Compose do not have that variant yet.

package com.kozmos.example

import androidx.compose.runtime.Composable
import androidx.compose.ui.platform.LocalContext
import com.kozmos.components.floorselector.KozmosFloorSelector
import com.kozmos.components.floorselector.KozmosFloorSelectorVariant
import com.kozmos.contracts.KozmosFloorPresentation

@Composable
fun Levels(selected: String, onSelect: (String) -> Unit) {
    KozmosFloorSelector(
        floors = listOf("B2", "B1", "G", "1", "2"),
        selectedFloor = selected,
        onFloorSelect = onSelect
    )
}

// The stepper's two buttons carry the product's names.
@Composable
fun LevelStepper(selected: String, onSelect: (String) -> Unit) {
    KozmosFloorSelector(
        floors = listOf("2", "1", "G", "B1", "B2"),
        selectedFloor = selected,
        onFloorSelect = onSelect,
        variant = KozmosFloorSelectorVariant.CompactStepper,
        previousFloorLabel = "Floor up",
        nextFloorLabel = "Floor down"
    )
}

// The levels that hold results carry their count, said in the product's
// words: a count needs a plural rule, and the product has the locale.
@Composable
fun LevelsWithResults(selected: String, onSelect: (String) -> Unit) {
    val resources = LocalContext.current.resources
    KozmosFloorSelector(
        floors = listOf(
            KozmosFloorPresentation(id = "L2", label = "Level 2", shortLabel = "L2", resultCount = 3),
            KozmosFloorPresentation(id = "L1", label = "Level 1", shortLabel = "L1"),
            KozmosFloorPresentation(id = "G", label = "Ground", shortLabel = "G", resultCount = 0)
        ),
        selectedFloor = selected,
        onFloorSelect = onSelect,
        resultCountLabel = { count -> resources.getQuantityString(R.plurals.results, count, count) }
    )
}
    `}
