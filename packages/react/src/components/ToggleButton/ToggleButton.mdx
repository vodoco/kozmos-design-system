# ToggleButton

A button that maintains a pressed/unpressed choice, with default or outline styling.
Use `pressed` and `onPressedChange` for controlled state, or `defaultPressed` for an initial uncontrolled value.

import SwiftUI
import Kozmos

// The label is required, and it is what VoiceOver says: there is no
// separate name. The state is yours to hold, and VoiceOver hears the button
// as selected while it is on.
struct ShowLabelsToggle: View {
    @Binding var showLabels: Bool

    var body: some View {
        KozmosToggleButton(isOn: $showLabels, label: "Labels")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.togglebutton.KozmosToggleButton

// The label is what TalkBack says: there is no separate name. The state is
// yours to hold. An icon is an ImageVector, and an icon-only button has no
// name at all, as the icon is drawn without a description.
@Composable
fun ShowLabelsToggle(showLabels: Boolean, onShowLabelsChange: (Boolean) -> Unit) {
    KozmosToggleButton(
        checked = showLabels,
        onCheckedChange = onShowLabelsChange,
        label = "Labels"
    )
}
`}
