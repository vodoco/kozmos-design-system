# FloatingActionButton

A round, elevated Button for a prominent action, with a plus icon by default.
It stays inline unless `placement="fixed"` is selected; give an icon-only action a meaningful accessible name.

    swift={`
import SwiftUI
import Kozmos

// A plus by default: iconName is an SF Symbol's name. The button takes no
// label: VoiceOver says the one you give it, as aria-label is on the web, or
// else the symbol's own name. It sits where you put it: there is no fixed
// placement.
struct AddPlaceButton: View {
    let addPlace: () -> Void

    var body: some View {
        KozmosFloatingActionButton(action: addPlace)
            .accessibilityLabel("Add place")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.floatingactionbutton.KozmosFloatingActionButton

// A plus by default: icon is an ImageVector. The part takes no name: TalkBack
// calls it "Action", in English, whatever it does, and a contentDescription
// you add is read before that word, not instead of it. It sits where you put
// it: there is no fixed placement.
@Composable
fun AddPlaceButton(addPlace: () -> Unit) {
    KozmosFloatingActionButton(onClick = addPlace)
}
`}
