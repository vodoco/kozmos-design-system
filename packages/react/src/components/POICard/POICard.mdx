# POICard

A general-purpose place card with optional imagery, subtitle, description, badges and actions.
Its selectable identity and action footer are separate; use POIResultCard for SDK search-result state and POIDetailPanel for the full detail view.

import SwiftUI
import Kozmos

// The line under the title is category, where React's is subtitle, and the
// description is content of your own. The card is 300pt wide, with no badges
// or selection, and its title is not a heading.
struct CafeCard: View {
    var body: some View {
        KozmosPOICard(
            title: "Atrium Café",
            category: "Ground floor",
            descriptionContent: {
                KozmosText("Coffee and refreshments by the main entrance.", tone: .muted)
            }
        )
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.poicard.KozmosPOICard
import com.kozmos.components.text.KozmosText

// The line under the title is category, where React's is subtitle, and the
// description is content of your own. The card is 300dp wide, with no badges
// or selection, and its title is not a heading. Compose's KozmosText takes no
// tone, so the description is not muted as React's is.
@Composable
fun CafeCard() {
    KozmosPOICard(
        title = "Atrium Café",
        category = "Ground floor",
        description = { KozmosText(text = "Coffee and refreshments by the main entrance.") }
    )
}
`}
