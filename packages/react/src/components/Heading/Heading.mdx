# Heading

Renders a semantic heading using the Kozmos level-based type scale.
`level` sets the visual size and default heading element; use `as` when the document hierarchy needs a different element.

import { Heading } from "@kozmos-ds/react";

    swift={`
import SwiftUI
import Kozmos

// A level-1 heading unless level says otherwise: the level sets its size and
// the heading level VoiceOver's rotor reads. Its text is a String, and there
// is no \`as\`: the level is both.
struct VenueHeading: View {
    var body: some View {
        KozmosHeading("Venue overview")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.heading.KozmosHeading

// A level-1 heading unless level says otherwise: the level sets its size.
// TalkBack says "heading" but not which level, as Compose's semantics have
// none. Its text is a String, and there is no \`as\`.
@Composable
fun VenueHeading() {
    KozmosHeading(text = "Venue overview")
}
`}
