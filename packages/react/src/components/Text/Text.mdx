# Text

Applies Kozmos typography, colour, alignment and optional truncation to body text.
Choose a semantic element with `as`, or use Heading for a heading with a level-based type scale.

## Implementation

<PlatformSnippets
    react={`
import { Text } from "@kozmos-ds/react";
export function Example() {
  return (
    <>
      <Text>Find your way to reception.</Text>
    </>
  );
}
    `}
    vue={`
<script setup>
import { KozmosText } from "@kozmos-ds/vue"
</script>

<template>
  <KozmosText />
</template>
    `}
    swift={`
import SwiftUI
import Kozmos

// Body text on the type ramp. style and tone name a step and a purpose,
// such as .footnote and .muted, rather than a font and a colour.
struct ReceptionHint: View {
    var body: some View {
        KozmosText("Find your way to reception.")
    }
}
`}
    kotlin={`
import androidx.compose.runtime.Composable
import com.kozmos.components.text.KozmosText

// Body text. Compose's KozmosText takes the words and a modifier, nothing
// else: it has no style, tone or truncation, and draws in the ambient style.
@Composable
fun ReceptionHint() {
    KozmosText(text = "Find your way to reception.")
}
`}
